One-time database setup

Run this SQL once in Supabase → SQL Editor, then tap Retry.

create table if not exists cb_config ( id text primary key, data jsonb, updated_at timestamptz default now() ); create table if not exists cb_users ( id text primary key, name text, color_idx int, created_at timestamptz default now() ); create table if not exists cb_transactions ( id text primary key, user_id text, to_user_id text, type text, amount numeric, cat text, description text, bank text, mode text, tx_date text, created_at timestamptz default now() ); alter table cb_config enable row level security; alter table cb_users enable row level security; alter table cb_transactions enable row level security; create policy "public_all" on cb_config for all using (true) with check (true); create policy "public_all" on cb_users for all using (true) with check (true); create policy "public_all" on cb_transactions for all using (true) with check (true);
